<script>
    // Function to detect if an element is in the viewport
    function isInViewport(element) {
        const rect = element.getBoundingClientRect();
        return rect.top < window.innerHeight && rect.bottom >= 0;
    }

    // Function to handle visibility of the rise-text
    function handleVisibility() {
        const riseText = document.querySelector('.rise-text');
        if (riseText && isInViewport(riseText)) {
            riseText.classList.add('visible');
        }
    }
//sidenav bar contact page

function openNav() {
    document.getElementById('sideNav').style.right = '0'; // Open from the right
}

function closeNav() {
    document.getElementById('sideNav').style.right = '-250px'; // Hide it back to the right
}



    // Attach event listeners
    document.addEventListener("scroll", handleVisibility);
    window.addEventListener("load", handleVisibility);
    window.addEventListener("resize", handleVisibility); // Ensure it works on viewport resize
</script>
